    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/stats-prefix.h" "#pragma once\n")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/stats-prefix.cpp" "#include \"stats-prefix.h\"\n")
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" /nologo /utf-8 /showIncludes /Zs "${CMAKE_CURRENT_BINARY_DIR}/stats-prefix.cpp"
        OUTPUT_VARIABLE _stats_includes ERROR_VARIABLE _stats_errors ENCODING UTF-8)
    string(APPEND _stats_includes "${_stats_errors}")
    if(_stats_includes MATCHES "注意: 包含文件:  ")
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
    elseif(_stats_includes MATCHES "Note: including file:  ")
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:  ")
    endif()
endif()

function(ay_stats_write_test_features)
    foreach(_feature batch assets gpu entity script scene host inspector editor)
        set(_stats_${_feature} OFF)
    endforeach()
    foreach(_pair "batch:AYStatsBatch" "assets:AYStatsAssets" "gpu:AYStatsGpu" "entity:AYStats_EntityTests" "script:AYStatsScriptIntegration" "scene:AYStatsSceneIntegration" "host:AYStats_HostTests" "inspector:AYStats_InspectorTests" "editor:AYStats_EditorTests")
        string(REPLACE ":" ";" _parts "${_pair}")
        list(GET _parts 0 _key)
        list(GET _parts 1 _target)
        if(TARGET ${_target})
            set(_stats_${_key} ON)
        endif()
    endforeach()
    set(_stats_network OFF)
    if(TARGET AYNetwork AND TARGET AYStats_EntityTests)
        set(_stats_network ON)
    endif()
    file(WRITE "${AY_STATS_BINARY_DIR}/unittest/test-features.json" "{\"batch\":\"${_stats_batch}\",\"assets\":\"${_stats_assets}\",\"gpu\":\"${_stats_gpu}\",\"entity\":\"${_stats_entity}\",\"script\":\"${_stats_script}\",\"network\":\"${_stats_network}\",\"scene\":\"${_stats_scene}\",\"host\":\"${_stats_host}\",\"inspector\":\"${_stats_inspector}\",\"editor\":\"${_stats_editor}\"}\n")
endfunction()
ay_stats_write_test_features()
function(ay_stats_add_resource_integration)
    if(NOT TARGET AYResource OR NOT TARGET AYStatsAssets)
        return()
    endif()
    add_library(AYStatsResources STATIC "${AY_STATS_SOURCE_DIR}/src/RecipeResource.cpp")
    add_library(AYStats::Resources ALIAS AYStatsResources)
    target_link_libraries(AYStatsResources PUBLIC AYStatsAssets AYResource)
    target_compile_definitions(AYResource PRIVATE AY_RESOURCE_HAS_STATS=1)
    if(TARGET AYResourceImport)
        target_sources(AYResourceImport PRIVATE "${AY_STATS_SOURCE_DIR}/src/StatsConverter.cpp")
        target_link_libraries(AYResourceImport PRIVATE AYStatsAssets)
        target_compile_definitions(AYResourceImport PRIVATE AY_RESOURCE_HAS_STATS=1)
    endif()
    if(TARGET AYProjectBuild)
        target_link_libraries(AYProjectBuild PRIVATE AYStatsAssets)
        target_compile_definitions(AYProjectBuild PRIVATE AY_RESOURCE_HAS_STATS=1)
    endif()
    if(TARGET AYResourceCook)
        target_link_libraries(AYResourceCook PRIVATE AYStatsAssets)
        target_compile_definitions(AYResourceCook PRIVATE AY_RESOURCE_HAS_STATS=1)
    endif()
    if(BUILD_TESTING AND TARGET AYProjectBuild AND TARGET AYResourceCook)
        add_executable(AYStats_ResourceTests "${AY_STATS_SOURCE_DIR}/unittest/RecipePackagingTests.cpp")
        target_link_libraries(AYStats_ResourceTests PRIVATE AYStatsResources AYProjectBuild AYResourceCook)
        target_compile_definitions(AYStats_ResourceTests PRIVATE AY_STATS_EXAMPLE_ASSETS="${AY_STATS_SOURCE_DIR}/demo/assets")
        add_test(NAME AYStats_ResourceTests COMMAND AYStats_ResourceTests)
        set_tests_properties(AYStats_ResourceTests PROPERTIES LABELS "stats;resource;integration" TIMEOUT 60 RUN_SERIAL TRUE)
    endif()
endfunction()
function(ay_stats_add_entity_integration)
    if(NOT TARGET AYEntityCore OR NOT TARGET AYStatsAssets OR NOT TARGET AYStatsBatch)
        message(FATAL_ERROR "Stats Entity integration requires EntityCore, StatsAssets and StatsBatch")
    endif()
    add_library(AYEntityStatsIntegration STATIC "${AY_STATS_SOURCE_DIR}/src/EntityIntegration.cpp")
    add_library(AYStats::Entity ALIAS AYEntityStatsIntegration)
    target_link_libraries(AYEntityStatsIntegration PUBLIC AYStatsAssets AYStatsBatch AYEntityCore)
    if(TARGET AYStatsResources)
        target_link_libraries(AYEntityStatsIntegration PUBLIC AYStatsResources)
        target_compile_definitions(AYEntityStatsIntegration PRIVATE AY_STATS_HAS_RESOURCES=1)
    endif()
    if(MSVC)
        target_compile_options(AYEntityStatsIntegration PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
    endif()
    add_executable(AYStats_GameplayExample EXCLUDE_FROM_ALL "${AY_STATS_SOURCE_DIR}/demo/GameplayExample.cpp")
    target_link_libraries(AYStats_GameplayExample PRIVATE AYEntityStatsIntegration)
    target_compile_definitions(AYStats_GameplayExample PRIVATE AY_STATS_EXAMPLE_ASSETS="${AY_STATS_SOURCE_DIR}/demo/assets")
    add_executable(AYStats_EntitySettlementBenchmark EXCLUDE_FROM_ALL "${AY_STATS_SOURCE_DIR}/demo/EntitySettlementBenchmark.cpp")
    target_link_libraries(AYStats_EntitySettlementBenchmark PRIVATE AYEntityStatsIntegration)
    if(BUILD_TESTING)
        add_executable(AYStats_EntityTests "${AY_STATS_SOURCE_DIR}/unittest/main.cpp" "${AY_STATS_SOURCE_DIR}/unittest/Test_Entity.cpp" "${AY_STATS_SOURCE_DIR}/unittest/Test_GameplayStats.cpp")
        target_link_libraries(AYStats_EntityTests PRIVATE AYEntityStatsIntegration AYTest)
        target_compile_definitions(AYStats_EntityTests PRIVATE AY_STATS_EXAMPLE_ASSETS="${AY_STATS_SOURCE_DIR}/demo/assets")
        ay_add_test(AYStats_EntityTests CASE_REPORT LABELS "stats;stats-integration;stats-full;integration" TIMEOUT 60 RUN_SERIAL)
    endif()
    ay_stats_write_test_features()
endfunction()
