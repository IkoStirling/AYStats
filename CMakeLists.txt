cmake_minimum_required(VERSION 3.25)
project(AYStats VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(PROJECT_IS_TOP_LEVEL AND MSVC)
    add_compile_options(/utf-8)
endif()
include(CTest)
if(PROJECT_IS_TOP_LEVEL AND MSVC AND CMAKE_GENERATOR MATCHES "Ninja")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/stats-prefix.h" "#pragma once\n")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/stats-prefix.cpp" "#include \"stats-prefix.h\"\n")
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" /nologo /utf-8 /showIncludes /Zs "${CMAKE_CURRENT_BINARY_DIR}/stats-prefix.cpp"
        OUTPUT_VARIABLE _stats_includes ERROR_VARIABLE _stats_errors ENCODING UTF-8)
    string(APPEND _stats_includes "${_stats_errors}")
    if(_stats_includes MATCHES "注意: 包含文件:  ")
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
    elseif(_stats_includes MATCHES "Note: including file:  ")
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:  ")
    endif()
endif()
add_library(AYStatsCore STATIC src/Stats.cpp src/Calculation.cpp src/Effects.cpp)
add_library(AYStats::Core ALIAS AYStatsCore)
target_compile_features(AYStatsCore PUBLIC cxx_std_23)
target_include_directories(AYStatsCore PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
if(MSVC)
    target_compile_options(AYStatsCore PRIVATE /W4 /utf-8 /Zc:__cplusplus)
endif()
option(AY_STATS_BUILD_BATCH "Build AYTask-backed CPU batch execution" ON)
if((BUILD_TESTING OR AY_STATS_BUILD_BATCH) AND NOT TARGET AYTask)
    if(NOT TARGET AYTest)
        add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../AYFoundation/AYTest ${CMAKE_CURRENT_BINARY_DIR}/foundation/test)
        add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../AYFoundation/AYPlatform ${CMAKE_CURRENT_BINARY_DIR}/foundation/platform)
    endif()
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../AYFoundation/AYTask ${CMAKE_CURRENT_BINARY_DIR}/foundation/task)
endif()
if(AY_STATS_BUILD_BATCH)
    add_library(AYStatsBatch STATIC src/Batch.cpp)
    add_library(AYStats::Batch ALIAS AYStatsBatch)
    target_link_libraries(AYStatsBatch PUBLIC AYStatsCore AYTask)
    add_executable(AYStats_Benchmark EXCLUDE_FROM_ALL demo/StatsBenchmark.cpp)
    target_link_libraries(AYStats_Benchmark PRIVATE AYStatsBatch)
endif()
if(BUILD_TESTING)
    add_subdirectory(unittest)
endif()
